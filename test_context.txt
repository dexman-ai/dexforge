/// Eigenvalues of a real upper-Hessenberg matrix via the bounded Francis
/// double-shift QR algorithm (real Schur decomposition).
///
/// `h` is consumed and overwritten with its (quasi-)triangular Schur form. The
/// returned array holds the `N` eigenvalues; complex eigenvalues appear as
/// conjugate pairs extracted from the 2x2 diagonal blocks of the Schur form.
///
/// The algorithm follows Golub & Van Loan, *Matrix Computations*, §7.5. It is a
/// fixed-size, stack-allocated port: there is no allocation and the outer loop
/// is capped at [`QR_MAX_ITERATIONS`].
pub fn hessenberg_eigenvalues<const N: usize>(mut h: [[f64; N]; N]) -> [Complex; N]

/// Roots of the real polynomial
/// `coeffs[0] x^4 + coeffs[1] x^3 + coeffs[2] x^2 + coeffs[3] x + coeffs[4]`.
///
/// Mirrors `numpy.roots`: leading near-zero coefficients reduce the effective
/// degree and trailing zero coefficients contribute roots at the origin. The
/// remaining roots are the eigenvalues of the (monic) companion matrix, found
/// with the bounded [`hessenberg_eigenvalues`] solver.
///
/// Returns the fixed-size root buffer together with the number of valid roots
/// (the effective polynomial degree). Unused slots are left as zero.
pub fn quartic_roots(coeffs: &[f64; 5]) -> ([Complex; 4], usize)

/// Orthonormal basis (14x6) of the left null space of the 14x8 matrix `q`.
///
/// Computed from the six smallest-eigenvalue eigenvectors of the symmetric
/// PSD Gram matrix `Q Qᵀ` (14x14). The particular orthonormal basis chosen
/// does not affect the downstream IK roots: any `N' = N O` with orthogonal
/// `O` yields the same `det M(x) = 0` roots and the same kernel `v_12`.
pub fn rr_left_null_space(q: &SMatrix<f64, 14, 8>) -> SMatrix<f64, 14, 6>

/// Moore-Penrose pseudo-inverse (8x14) of the 14x8 matrix `q`, via SVD.
///
/// Falls back to the all-zero matrix if the decomposition does not converge;
/// callers treat the resulting candidate as degenerate and skip it.
pub fn rr_pseudo_inverse(q: &SMatrix<f64, 14, 8>) -> SMatrix<f64, 8, 14>

/// Real `tan(q/2)` roots of `det M(x) = 0` and their kernels via the 24x24
/// companion linearisation (Manocha-Canny Theorem 1).
///
/// Builds the companion `Sigma = [[0, I], [-A⁻¹C, -A⁻¹B]]` from the
/// *equilibrated* pencil, reduces it to upper-Hessenberg form and extracts the
/// eigenvalues with the bounded Francis-QR [`hessenberg_eigenvalues`]. Spurious
/// near-`+/-i` roots and roots with non-negligible imaginary parts are
/// filtered out. For every accepted real root `x` the matching 12-vector kernel
/// `v_12` of `M(x)` is recovered (`v_12 = d_r ⊙ nullvec(M_eq(x))`).
///
/// Fills `roots`/`vecs` and returns the number of accepted roots (`<= 16`). If
/// the equilibrated leading matrix is singular, returns `0`.
///
/// `cond_out` receives the 2-norm condition number of the equilibrated leading
/// matrix `A_eq` (== Python's `np.linalg.cond(a_eq)`). Callers compare it to the
/// `1e10` threshold to detect the ill-conditioned poses that the Python
/// reference handles with its Möbius/random-eigenvalue fallback.
pub fn rr_real_roots(
    m_quad: &SMatrix<f64, 12, 12>,
    m_lin: &SMatrix<f64, 12, 12>,
    m_const: &SMatrix<f64, 12, 12>,
    roots: &mut [f64; 24],
    vecs: &mut [[f64; 12]; 24],
    cond_out: &mut f64,
) -> usize

/// Bounded Levenberg-Marquardt polish for a joint vector of **any** DOF.
///
/// This is the universal LM loop shared across all arm drivers.  The
/// arm-specific part — computing the 12-vector pose residual from the arm's
/// own FK — is supplied as a closure so the function remains free of per-arm
/// types and constants.  The joint-space dimension is a const generic
/// parameter `DOF`, so all matrix allocations (Jacobian `J : 12×DOF`,
/// normal-equations `JᵀJ : DOF×DOF`, step `dq : DOF×1`) live entirely on
/// the stack: no heap, no `Vec`, compatible with `#![no_std]`.
///
/// `residual_fn(q)` must return `[vec(R(q) − R_t); t(q) − t_t]` — the
/// standard 12-vector pose residual.  The closure typically captures the
/// target pose and delegates to the driver's local `pose_residual12`.
///
/// The const generic parameter is monomorphised at each call site, so the
/// compiled cost is identical to having the loop inlined in each driver.
///
/// Returns `Some(q_refined)` when the FK residual falls within `fk_atol`,
/// or `None` if the iteration budget is exhausted without convergence.
///
/// # Examples
///

///
/// // 7-DOF arm — all seven joints are refined simultaneously
/// lm_refine::<7, _>(&q7, |q| residual12_7dof(q, &target), max_iters, 1e-10);

/// // 8+ DOF gantry
/// lm_refine::<8, _>(&q8, |q| residual12_8dof(q, &target), max_iters, 1e-10);
/// ```

    max_iters: usize,
    fk_atol: f64,
) -> Option<[f64; DOF]>

    F: Fn(&[f64; DOF]) -> [f64; 12],

/// Bounded Levenberg-Marquardt polish that refines **only a subset of joints**.
///
/// This is the universal masked LM loop shared by every redundant 7R
/// `jointlock.seven_r` driver (Franka, xArm7, Rizon).  Those solvers collapse
/// a redundant arm to a 6R sub-problem by locking one joint to a sample value;
/// the polish must then refine the remaining free joints while **holding the
/// locked joint fixed**, otherwise the redundant arm would drift along its
/// self-motion manifold instead of converging onto the isolated root.
///
/// The set of free joints is supplied as `free` (a `[usize; FREE]` array of
/// joint indices into the `DOF`-vector); the locked joint is simply any index
/// absent from `free`.  Both the full joint count `DOF` and the free-joint
/// count `FREE` are const generic parameters, so every matrix
/// (Jacobian `J : 12×FREE`, normal equations `JᵀJ : FREE×FREE`, step
/// `dq : FREE×1`) lives entirely on the stack — no heap, `#![no_std]`-safe,
/// and hard-bounded by `max_iters`.
///
/// `residual_fn(q)` must return the standard 12-vector pose residual
/// `[vec(R(q) − R_t); t(q) − t_t]` of the **full** `DOF`-joint FK.
///
/// Unlike [`lm_refine`], this returns the refined joint vector unconditionally
/// (the redundant-arm drivers FK-verify and de-duplicate the result
/// downstream, so there is no convergence/`fk_atol` gate here).
///
/// # Examples
///
/// ```ignore
/// // 7R arm with joint 4 locked: refine joints {0,1,2,3,5,6}.
/// const FREE: [usize; 6] = [0, 1, 2, 3, 5, 6];
/// let q = lm_refine_masked::<7, 6, _>(&q7, &FREE, |q| residual12(q, &target), max_iters);
/// ```
pub fn lm_refine_masked<const DOF: usize, const FREE: usize, F>(
    q_in: &[f64; DOF],
    free: &[usize; FREE],
    residual_fn: F,
    max_iters: usize,
) -> [f64; DOF]
where
    F: Fn(&[f64; DOF]) -> [f64; 12],

/// Dampened Least Squares (DLS) IK refinement for a **calibrated** arm with
/// **any** DOF.
///
/// Takes an initial joint configuration `q_init` (e.g., the analytical IK
/// solution computed for the *idealized* arm model) and the actual arm's
/// forward-kinematics function `fk_fn` (which may differ from the idealized
/// model due to calibration offsets, manufacturing tolerances, or a different
/// URDF), and iteratively refines the joint angles until the FK of the actual
/// arm satisfies the pose goal within `fk_atol`.
///
/// Internally delegates to [`lm_refine`] with the standard 12-vector pose
/// residual `[vec(R(q) − R_goal); t(q) − t_goal]`, so all coding restrictions
/// are inherited: zero allocation, `#![no_std]`-compatible, and bounded
/// iteration count (hard-capped at `max_iters`).
///
/// # Arguments
///
/// * `goal`      — Target end-effector pose as a [`Transform`].
/// * `q_init`    — Initial joint configuration (radians). Typically the
///                 analytical IK solution for the idealized arm.
/// * `fk_fn`     — Forward kinematics of the **actual** arm.  Must be
///                 deterministic and return a [`Transform`].
/// * `max_iters` — Hard iteration ceiling. Use [`DLS_MAX_ITERS`] unless a
///                 tighter real-time budget is needed.
/// * `fk_atol`   — FK closure tolerance (Frobenius norm). Use [`DLS_FK_ATOL`]
///                 (`1e-5`) to match the rest of the library.
///
/// # Returns
///
/// `Some(q_refined)` when `frob_diff(fk_fn(q_refined), goal) ≤ fk_atol`,
/// `None` if the iteration budget is exhausted without convergence.
///
/// # Examples
///
/// ```ignore
/// // Refine a 6-DOF analytical seed onto a calibrated arm.
/// let q_seed: [f64; 6] = analytical_ik(&idealized_model, &goal);

    a single native-code chunk under Cython compile.

    Bottom row of the accumulator stays [0, 0, 0, 1] implicitly.
    """
    n = len(_JOINT_AXES)
    # Identity accumulator (the bottom row [0,0,0,1] is implicit).
    a00 = 1.0; a01 = 0.0; a02 = 0.0; a03 = 0.0
    a10 = 0.0; a11 = 1.0; a12 = 0.0; a13 = 0.0
    a20 = 0.0; a21 = 0.0; a22 = 1.0; a23 = 0.0
    for i in range(n):
        # Inline Rodrigues for this joint's axis.
        ax = float(_JOINT_AXES[i][0])
        ay = float(_JOINT_AXES[i][1])
        az = float(_JOINT_AXES[i][2])
        qi = float(q[i])
        c = math.cos(qi); s = math.sin(qi); oc = 1.0 - c
        r00 = c + ax*ax*oc;     r01 = ax*ay*oc - az*s; r02 = ax*az*oc + ay*s
        r10 = ay*ax*oc + az*s;  r11 = c + ay*ay*oc;    r12 = ay*az*oc - ax*s
        r20 = az*ax*oc - ay*s;  r21 = az*ay*oc + ax*s; r22 = c + az*az*oc
        # T_left[i] entries.
        Tl = _JOINT_T_LEFTS[i]
        l00 = float(Tl[0,0]); l01 = float(Tl[0,1])
        l02 = float(Tl[0,2]); l03 = float(Tl[0,3])
        l10 = float(Tl[1,0]); l11 = float(Tl[1,1])
        l12 = float(Tl[1,2]); l13 = float(Tl[1,3])
        l20 = float(Tl[2,0]); l21 = float(Tl[2,1])
        l22 = float(Tl[2,2]); l23 = float(Tl[2,3])
        # M = T_left[i] @ R (R is the homogeneous version of the 3x3
        # rotation above with column 3 = [0,0,0,1]^T).
        m00 = l00*r00 + l01*r10 + l02*r20
        m01 = l00*r01 + l01*r11 + l02*r21
        m02 = l00*r02 + l01*r12 + l02*r22
        m03 = l03
        m10 = l10*r00 + l11*r10 + l12*r20
        m11 = l10*r01 + l11*r11 + l12*r21
        m12 = l10*r02 + l11*r12 + l12*r22
        m13 = l13
        m20 = l20*r00 + l21*r10 + l22*r20
        m21 = l20*r01 + l21*r11 + l22*r21
        m22 = l20*r02 + l21*r12 + l22*r22
        m23 = l23
        # T_right[i] entries.
        Tr = _JOINT_T_RIGHTS[i]
        t00 = float(Tr[0,0]); t01 = float(Tr[0,1])
        t02 = float(Tr[0,2]); t03 = float(Tr[0,3])
        t10 = float(Tr[1,0]); t11 = float(Tr[1,1])
        t12 = float(Tr[1,2]); t13 = float(Tr[1,3])
        t20 = float(Tr[2,0]); t21 = float(Tr[2,1])
        t22 = float(Tr[2,2]); t23 = float(Tr[2,3])
        # N = M @ T_right[i]
        n00 = m00*t00 + m01*t10 + m02*t20
        n01 = m00*t01 + m01*t11 + m02*t21
        n02 = m00*t02 + m01*t12 + m02*t22
        n03 = m00*t03 + m01*t13 + m02*t23 + m03
        n10 = m10*t00 + m11*t10 + m12*t20
        n11 = m10*t01 + m11*t11 + m12*t21
        n12 = m10*t02 + m11*t12 + m12*t22
        n13 = m10*t03 + m11*t13 + m12*t23 + m13
        n20 = m20*t00 + m21*t10 + m22*t20
        n21 = m20*t01 + m21*t11 + m22*t21
        n22 = m20*t02 + m21*t12 + m22*t22
        n23 = m20*t03 + m21*t13 + m22*t23 + m23
        # T_acc = T_acc @ N
        b00 = a00*n00 + a01*n10 + a02*n20
        b01 = a00*n01 + a01*n11 + a02*n21
        b02 = a00*n02 + a01*n12 + a02*n22
        b03 = a00*n03 + a01*n13 + a02*n23 + a03
        b10 = a10*n00 + a11*n10 + a12*n20
        b11 = a10*n01 + a11*n11 + a12*n21
        b12 = a10*n02 + a11*n12 + a12*n22
        b13 = a10*n03 + a11*n13 + a12*n23 + a13
        b20 = a20*n00 + a21*n10 + a22*n20
        b21 = a20*n01 + a21*n11 + a22*n21
        b22 = a20*n02 + a21*n12 + a22*n22
        b23 = a20*n03 + a21*n13 + a22*n23 + a23
        a00, a01, a02, a03 = b00, b01, b02, b03
        a10, a11, a12, a13 = b10, b11, b12, b13
        a20, a21, a22, a23 = b20, b21, b22, b23
    return np.array(
        [[a00, a01, a02, a03],
         [a10, a11, a12, a13],
         [a20, a21, a22, a23],
         [0.0, 0.0, 0.0, 1.0]],
        dtype=np.float64,
    )


@cython.ccall
@cython.locals(i=cython.int, n=cython.int)
def _spatial_jacobian(q):
    """6 x n_dof spatial Jacobian using the baked chain constants.

    Math identical to ssik.refinement.kinbody_jacobian: column i
    is (p_i x z_i, z_i) where z_i is the i-th joint axis in the
    world frame at q and p_i is the i-th joint origin. This is
    the SPATIAL twist representation -- T(q+dq) @ T(q)^-1 ~
    exp([J @ dq]) -- matching the residual extracted by
    ssik.refinement.se3_log_residual. Per-arm version with
    baked _JOINT_AXES / _JOINT_T_LEFTS / _JOINT_T_RIGHTS so
    there's no KinBody walk at runtime.
    """
    n = len(_JOINT_AXES)
    cum = _FK_EYE4.copy()
    cums = [cum.copy()]
    rot = _FK_EYE4.copy()
    for i in range(n):
        rot[:3, :3] = _rotation_matrix(_JOINT_AXES[i], float(q[i]))
        cum = cum @ _JOINT_T_LEFTS[i] @ rot @ _JOINT_T_RIGHTS[i]
        cums.append(cum.copy())
    J = np.zeros((6, n), dtype=np.float64)
    for i in range(n):
        t_pre = cums[i] @ _JOINT_T_LEFTS[i]
        axis_unit = _JOINT_AXES[i] / np.linalg.norm(_JOINT_AXES[i])
        z_i = t_pre[:3, :3] @ axis_unit
        p_i = t_pre[:3, 3]
        J[:3, i] = np.cross(p_i, z_i)
        J[3:, i] = z_i
    return J


@cython.ccall
def _wrap_to_pi(a: float) -> float:
    """Wrap an angle to ``(-pi, pi]``. Called inside the per-IK
    dedup hot loop (235k+ times on Franka 7R)."""
    return ((a + math.pi) % _TWO_PI) - math.pi


@cython.ccall
@cython.locals(
    i=cython.int,
    n=cython.int,
    diff=cython.double,
    ai=cython.double,
    bi=cython.double,
)
def _q_close_wrap(a, b, tol: float) -> bool:
    """Return ``True`` if joint vectors ``a`` and ``b`` agree (mod 2pi)
    within ``tol`` per element. Replaces the
    ``np.array([_wrap_to_pi(...)]) -> np.all(np.abs(...) < tol)``
    pipeline that allocated a numpy array per dedup-loop iteration --
    a per-element scalar loop avoids the array creation and the
    ``np.all`` reduction overhead, which together dominated the
    artifact's ``solve()`` body at the per-IK level."""
    n = len(a)
    for i in range(n):
        ai = float(a[i])
        bi = float(b[i])
        diff = ((ai - bi + math.pi) % _TWO_PI) - math.pi
        if abs(diff) > tol:
            return False
    return True


def solve(
    T_target,
    *,
    max_solutions: int | None = None,
    q_seed=None,
    respect_limits: bool = True,
    allow_refinement: bool = False,
    policy: TolerancePolicy = DEFAULT_TOLERANCE_POLICY,
    refinement_max_iters: int = 15,
):
    """Inverse kinematics. Returns ``list[Solution]``.

    :param T_target: 4x4 SE(3) target end-effector pose.
    :param max_solutions: optional cap on returned IKs (post-dedup,
        post-limits filter). ``None`` = full redundancy enumeration.
        Combine with ``q_seed`` for the "give me the IK closest to
        where I am now" trajectory-tracking idiom.
    :param q_seed: optional joint configuration. When provided,
        returned solutions are sorted by wrap-to-pi distance from
        ``q_seed`` (closest first). Has no effect on which solutions
        are returned, only their order.
    :param respect_limits: when ``True`` (default), solutions
        outside URDF joint limits are dropped. Pass ``False`` for
        the raw geometric set (e.g. analysis / debugging).
    :param allow_refinement: opt into Newton polish for near-miss
        algebraic candidates that don't quite meet ``fk_atol``.
        Default ``False`` -- the algebraic path is already at
        machine precision on tier-0 / SRS arms. On tier-2 RR
        arms (JACO 2, Rizon 4, Kassow), polish can recover
        edge-case candidates whose algebraic FK drifts above
        ``fk_atol``, at ~100-300 us per polished branch.
    :param policy: tolerance policy (FK closure + dedup tolerance).
        Rarely customised.
    :param refinement_max_iters: cap on Newton iterations per
        candidate when ``allow_refinement=True``.
    :returns: list of :class:`Solution`; empty list iff no IK
        closed within ``policy.subproblem_numerical`` (or all
        IKs were filtered by ``respect_limits=True``).
    """
    T = np.asarray(T_target, dtype=np.float64)
    candidates = _solve_algebraic(T)

    fk_atol = policy.subproblem_numerical
    dedup_atol = policy.subproblem_dedup

    # Three-bucket sort: exact (closes within fk_atol), near-miss
    # (refinable when allow_refinement=True), or drop.
    verified: list[tuple[np.ndarray, float, str, int]] = []
    for cand_q in candidates:
        q = np.asarray(cand_q, dtype=np.float64)
        if not np.all(np.isfinite(q)):
            continue
        T_check = _fk(q)
        residual = float(np.linalg.norm(T_check - T))
        if residual <= fk_atol:
            verified.append((q, residual, "none", 0))
            continue
        if not allow_refinement:
            continue
        # Newton polish using the per-arm spatial Jacobian.
        refined = _lm_refine(
            q,
            _fk,
            T,
            fk_atol=fk_atol,
            max_iters=refinement_max_iters,
            jacobian_fn=_spatial_jacobian,
        )
        if refined is None:
            continue
        q_ref, resid_ref, iters = refined
        verified.append((q_ref, resid_ref, "lm", iters))

    # Wrap-to-pi dedup; keep lowest fk_residual on collision.
    # Inner check via ``_q_close_wrap`` -- typed scalar loop, no per-
    # iteration numpy allocation (#137 Slice 3).
    deduped: list[tuple[np.ndarray, float, str, int]] = []
    for cand_q, cand_res, ref_used, ref_iters in verified:
        dup_idx = None
        for j, (existing_q, _, _, _) in enumerate(deduped):
            if _q_close_wrap(cand_q, existing_q, dedup_atol):
                dup_idx = j
                break
        if dup_idx is None:
            deduped.append((cand_q, cand_res, ref_used, ref_iters))
        elif cand_res < deduped[dup_idx][1]:
            deduped[dup_idx] = (cand_q, cand_res, ref_used, ref_iters)

    solutions = [
        Solution(
            q=q,
            fk_residual=residual,
            refinement_used=ref_used,
        )
        for q, residual, ref_used, _ref_iters in deduped
    ]

    # Post-processing pass (#238 item 4). Order matters:
    #   1. wrap_to_limits tries q +/- 2*pi per joint to bring
    #      candidates into the URDF's limit range
    #   2. respect_limits drops anything still outside
    #   3. nearest_to_seed sorts by distance to q_seed (if given)
    #   4. max_solutions truncates to the first k
    if respect_limits:
        solutions = _ps_wrap_to_limits(solutions, _KB)
        solutions = _ps_respect_limits(solutions, _KB)
    if q_seed is not None:
        solutions = _ps_nearest_to_seed(solutions, q_seed)
    if max_solutions is not None and len(solutions) > max_solutions:
        solutions = solutions[:max_solutions]
    return solutions

fk = _fk

//! Analytical inverse kinematics for the Universal Robots **UR5**.
//!
//! This is a line-by-line translation of the Python `ssik` prebuilt artifact
//! `ssik/prebuilt/ur5_ik.py` (solver `ikgeo.three_parallel`, tier 0) into a
//! deterministic, `#![no_std]`, zero-allocation Rust implementation.
//!
//! The closed form is: subproblem **SP6** for the coupled wrist/shoulder angle
//! pair `(q1, q5)`, followed by the inlined `SP1 + SP3 + SP1` chain for the
//! remaining joints. SP6 reduces to intersecting two conics, which produces a
//! quartic solved through the companion-matrix eigenvalue routine in
//! [`crate::math::ssik_core`].
//!
//! ## Equivalence to the Python reference
//!
//! [`solve`] mirrors `ur5_ik.solve` with the same argument list:
//! `max_solutions`, `q_seed`, `respect_limits`, `allow_refinement`, `policy`,
//! and `refinement_max_iters`. The algebraic solution set is FK-verified to
//! [`TolerancePolicy::subproblem_numerical`] and de-duplicated modulo `2*pi`.
//!
//! [`fk`] mirrors `ur5_ik.fk` (alias of `_fk`): product-of-exponentials forward
//! kinematics using the baked chain constants.

use nalgebra::SMatrix;

use crate::math::kinematics::{lm_refine6, mat4_frob_diff};
use crate::math::ssik_core::{sp6_solve, wrap_to_pi};
pub use crate::math::tolerances::TolerancePolicy;

use core::f64::consts::PI;

extern crate std;

/// Degrees of freedom of the UR5.
pub const DOF: usize = 6;

// --- algebraic-block degeneracy constants from ur5_ik.py ---
const TWO_PI: f64 = 2.0 * PI;

/// Joint angle limits (lo, hi) in radians — baked from `_JOINT_LIMITS` in
/// the Python artifact.  Used by [`solve`] when `respect_limits = true`.
const JOINT_LIMITS: [(f64, f64); 6] = [
    (-6.28318530717959, 6.28318530717959),
    (-6.28318530717959, 6.28318530717959),
    (-3.14159265358979, 3.14159265358979),
    (-6.28318530717959, 6.28318530717959),
    (-6.28318530717959, 6.28318530717959),
    (-6.28318530717959, 6.28318530717959),
];

/// Inverse kinematics — mirrors `ur5_ik.solve`.
///
/// # Arguments
///
/// * `t` — 4×4 SE(3) target pose of `ee_link` in `base_link`.
/// * `max_solutions` — cap on returned solutions (post-dedup, post-limits).
///   `None` returns the full algebraic set (at most 8 for the UR5).
/// * `q_seed` — when `Some`, solutions are sorted by wrap-to-pi distance
///   from the seed (closest first). Has no effect on *which* solutions are
///   returned.
/// * `respect_limits` — when `true`, solutions outside the baked URDF joint
///   limits are dropped (after a `±2π` wrap-into-limits attempt). Pass
///   `false` for the raw geometric set.
/// * `allow_refinement` — when `true`, algebraic candidates that exceed
///   `policy.subproblem_numerical` are polished with a bounded Levenberg-Marquardt step.
///   For the UR5 (tier-0 arm) this is rarely needed; default `false`.
/// * `policy` — FK closure and dedup tolerances; use
///   [`TolerancePolicy::DEFAULT`] for the Python defaults.
/// * `refinement_max_iters` — maximum LM iterations when
///   `allow_refinement = true`.
///
/// Returns up to 16 candidate joint vectors `[q1, .., q6]` in radians;
/// unused slots are [`None`].
pub fn solve(
    t: &SMatrix<f64, 4, 4>,
    max_solutions: Option<usize>,
    q_seed: Option<&[f64; 6]>,
    respect_limits: bool,
    allow_refinement: bool,
    policy: TolerancePolicy,
    refinement_max_iters: usize,
) -> [Option<[f64; 6]>; 16] {
    let fk_atol = policy.subproblem_numerical;
    let dedup_atol = policy.subproblem_dedup;

    let mut solutions: [Option<[f64; 6]>; 16] = [None; 16];
    let mut solution_count = 0usize;

    // Raw algebraic candidates (stack array, at most 8 for the UR5).
    let mut cands = [[0.0f64; 6]; 8];
    let cand_count = solve_algebraic(t, &mut cands, policy);

    for ci in 0..cand_count {
        let q = cands[ci];
        // Reject non-finite candidates (mirrors np.isfinite filter).
        if !q.iter().all(|v| v.is_finite()) {
            continue;
        }
        // FK closure check.
        let residual = fk_residual(&q, t);
        let accepted_q: [f64; 6];
        if residual <= fk_atol {
            accepted_q = q;
        } else if allow_refinement {
            // Newton polish for near-miss candidates, mirroring Python's
            // `allow_refinement` path (lm_refine via _spatial_jacobian).
            match refine6(&q, t, refinement_max_iters, fk_atol) {
                Some(refined) => {
                    accepted_q = refined;
                }
                None => continue,
            }
        } else {
            continue;
        }

        // Wrap-to-pi de-duplication against accepted solutions.
        let mut dup = false;
        for s in solutions.iter().take(solution_count) {
            if let Some(existing) = s {
                if q_close_wrap(&accepted_q, existing, dedup_atol) {
                    dup = true;
                    break;
                }
            }
        }
        if dup {
            continue;
        }
        if solution_count < 16 {
            solutions[solution_count] = Some(accepted_q);
            solution_count += 1;
        }
    }

    // Post-processing (mirrors Python's _ps_wrap_to_limits + _ps_respect_limits).
    if respect_limits {
        let mut filtered: [Option<[f64; 6]>; 16] = [None; 16];
        let mut count = 0usize;
        for sol in solutions.iter().take(solution_count).flatten() {
            let q = wrap_to_limits(sol);
            if in_limits(&q) && count < 16 {
                filtered[count] = Some(q);
                count += 1;
            }
        }
        solutions = filtered;
        solution_count = count;
    }

    // Sort by seed distance (mirrors Python's _ps_nearest_to_seed).
    if let Some(seed) = q_seed {
        // Insertion sort on the first `solution_count` slots.
        for i in 1..solution_count {
            let mut j = i;
            while j > 0 {
                let prev = solutions[j - 1].unwrap();
                let curr = solutions[j].unwrap();
                if seed_dist(&prev, seed) > seed_dist(&curr, seed) {
                    solutions.swap(j - 1, j);
                    j -= 1;
                } else {
                    break;
                }
            }
        }
    }

    // Truncate to max_solutions (mirrors Python's `solutions[:max_solutions]`).
    if let Some(max) = max_solutions {
        let cap = max.min(solution_count);
        for i in cap..solution_count {
            solutions[i] = None;
        }
    }

    solutions
}

/// `true` if joint vectors `a` and `b` agree (mod `2*pi`) within `tol` per axis.
fn q_close_wrap(a: &[f64; 6], b: &[f64; 6], tol: f64) -> bool {
    for i in 0..6 {
        if libm::fabs(wrap_to_pi(a[i] - b[i])) > tol {
            return false;
        }
    }
    true
}

/// Try `q[i] ± 2π` per joint to bring the vector into `JOINT_LIMITS`.
/// Mirrors Python's `_ps_wrap_to_limits`.
fn wrap_to_limits(q: &[f64; 6]) -> [f64; 6] {
    let mut out = *q;
    for i in 0..6 {
        let (lo, hi) = JOINT_LIMITS[i];
        if out[i] < lo {
            let wrapped = out[i] + TWO_PI;
            if wrapped <= hi {
                out[i] = wrapped;
            }
        } else if out[i] > hi {
            let wrapped = out[i] - TWO_PI;
            if wrapped >= lo {
                out[i] = wrapped;
            }
        }
    }
    out
}

/// `true` if every joint is within its URDF limit.
fn in_limits(q: &[f64; 6]) -> bool {
    for i in 0..6 {
        let (lo, hi) = JOINT_LIMITS[i];
        if q[i] < lo || q[i] > hi {
            return false;
        }
    }
    true
}

/// Squared wrap-to-π distance between `q` and `seed`. Used for seed sorting.
fn seed_dist(q: &[f64; 6], seed: &[f64; 6]) -> f64 {
    let mut s = 0.0;
    for i in 0..6 {
        let d = wrap_to_pi(q[i] - seed[i]);
        s += d * d;
    }
    s
}

/// 12-vector pose residual `[vec(R(q) − R_t); t(q) − t_t]` for the UR5.
fn pose_residual12(q: &[f64; 6], target: &SMatrix<f64, 4, 4>) -> [f64; 12] {
    let tf = fk(q);
    let mut r = [0.0f64; 12];
    let mut k = 0usize;
    for i in 0..3 {
        for j in 0..3 {
            r[k] = tf[(i, j)] - target[(i, j)];
            k += 1;
        }
    }
    for i in 0..3 {
        r[9 + i] = tf[(i, 3)] - target[(i, 3)];
    }
    r
}

/// Levenberg-Marquardt polish of a 6-DOF UR5 configuration onto the exact
/// pose root. Mirrors the Python `_lm_refine` path used when
/// `allow_refinement = True`. Returns `Some(q_refined)` if the FK residual
/// converges within `fk_atol`, otherwise `None`.
fn refine6(
    q_in: &[f64; 6],
    target: &SMatrix<f64, 4, 4>,
    max_iters: usize,
    fk_atol: f64,
) -> Option<[f64; 6]> {
    lm_refine6(q_in, |q| pose_residual12(q, target), max_iters, fk_atol)
}

/// Fills `out` with the raw algebraic IK candidates and returns the count.
fn solve_algebraic(t: &SMatrix<f64, 4, 4>, out: &mut [[f64; 6]; 8], policy: TolerancePolicy) -> usize {
    let r_00 = t[(0, 0)];
    let r_01 = t[(0, 1)];
    let r_02 = t[(0, 2)];
    let r_10 = t[(1, 0)];
    let r_11 = t[(1, 1)];
    let r_12 = t[(1, 2)];
    let r_20 = t[(2, 0)];
    let r_21 = t[(2, 1)];
    let r_22 = t[(2, 2)];
    let p_x = t[(0, 3)];
    let p_y = t[(1, 3)];
    let p_z = t[(2, 3)];

    let mut count = 0usize;

    // T_target-derived SP6 inputs.
    let p_16_x = p_x - 1.64748849439063e-19 * r_01 - 0.0823 * r_02;
    let p_16_y = p_y - 1.64748849439063e-19 * r_11 - 0.0823 * r_12;
    let p_16_z = p_z - 1.64748849439063e-19 * r_21 - 0.0823 * r_22;
    let r_06_axes5 = [1.0 * r_02, 1.0 * r_12, 1.0 * r_22];

    const H_SP_0: [f64; 3] = [0.0, -1.0, 6.123233995736766e-17];
    const K_SP_0: [f64; 3] = [-0.0, -0.0, -1.0];
    const K_SP_1: [f64; 3] = [0.0, -1.2246467991473532e-16, -1.0];
    const NEG_P_5: [f64; 3] = [-0.0, 1.3877787807814457e-17, 0.09465];
    const NEG_AXES_5: [f64; 3] = [-0.0, 1.0, -6.123233995736766e-17];

    let p_16 = [p_16_x, p_16_y, p_16_z];
    let h_sp = [H_SP_0, H_SP_0, H_SP_0, H_SP_0];
    let k_sp = [K_SP_0, K_SP_1, K_SP_0, K_SP_1];
    let p_sp = [p_16, NEG_P_5, r_06_axes5, NEG_AXES_5];

    let mut theta15 = [(0.0f64, 0.0f64); 4];
    let n15 = sp6_solve(&h_sp, &k_sp, &p_sp, 0.10915, 0.0, policy.subproblem_numerical, policy.subproblem_degeneracy, &mut theta15);

    for si in 0..n15 {
        let (q1, q5) = theta15[si];

        // NOTE: the literal constants in the theta14 / q6 / d_inner blocks
        // below (e.g. `6.12323399573677e-17`, `7.49879891330929e-33`) are the
        // machine-generated symbolic coefficients emitted by `ssik build` and
        // are copied verbatim from `ur5_ik.py`. Most are rounding residue of
        // exact zeros (cos(pi/2)) preserved here so the floating-point
        // accumulation order matches the Python reference bit-for-bit.

        // SP1 for theta14 = q1+q2+q3+q4.
        let th14_x0 = libm::sin(q5);
        let th14_x1 = 1.0 * r_22;
        let th14_x2 = libm::sin(q1);
        let th14_x3 = 6.12323399573677e-17 * r_01 - 1.0 * r_02;
        let th14_x4 = th14_x2 * th14_x3;
        let th14_x5 = 1.0 * r_01 + 6.12323399573677e-17 * r_02;
        let th14_x6 = 6.12323399573677e-17 * th14_x2 * th14_x5;
        let th14_x7 = libm::cos(q1);
        let th14_x8 = 6.12323399573677e-17 * r_11 - 1.0 * r_12;
        let th14_x9 = th14_x7 * th14_x8;
        let th14_x10 = 1.0 * r_11 + 6.12323399573677e-17 * r_12;
        let th14_x11 = 6.12323399573677e-17 * th14_x10 * th14_x7;
        let th14_x12 = th14_x11 + th14_x4 - th14_x6 - th14_x9;
        let th14_x13 = libm::cos(q5);
        let th14_x14 = 6.12323399573677e-17 * th14_x10 * th14_x2 - th14_x2 * th14_x8
            - th14_x3 * th14_x7
            + 6.12323399573677e-17 * th14_x5 * th14_x7;
        let th14_x15 = 1.0 * th14_x13;
        let theta14 = libm::atan2(
            -th14_x0 * th14_x1 - 6.12323399573677e-17 * th14_x0 * th14_x12
                + th14_x14 * (6.12323399573677e-17 - 6.12323399573677e-17 * th14_x13),
            -1.0 * th14_x0 * th14_x14
                + th14_x1 * (1.22464679914735e-16 * th14_x13 - 6.12323399573677e-17)
                + th14_x12 * (-th14_x15 - 7.49879891330929e-33)
                - (th14_x15 + 3.74939945665464e-33)
                    * (6.12323399573677e-17 * r_22 - th14_x11 - th14_x4 + th14_x6 + th14_x9),
        );

        // SP1 for q6 (wrist roll-2).
        let q6_x0 = libm::cos(q5);
        let q6_x1 = libm::sin(q1);
        let q6_x2 = 1.0 * q6_x1;
        let q6_x3 = libm::cos(q1);
        let q6_x4 = 1.0 * q6_x3;
        let q6_x5 = q6_x2 * r_00 - q6_x4 * r_10 + 6.12323399573677e-17 * r_20;
        let q6_x6 = libm::sin(q5);
        let q6_x7 = 6.12323399573677e-17 * r_22;
        let q6_x8 = 6.12323399573677e-17 * r_01 - 1.0 * r_02;
        let q6_x9 = q6_x4 * (6.12323399573677e-17 * r_11 - 1.0 * r_12);
        let q6_x10 = q6_x2 * q6_x8 - q6_x7 - q6_x9 + 3.74939945665464e-33 * r_21;
        let q6_x11 = 1.0 * r_01 + 6.12323399573677e-17 * r_02;
        let q6_x12 = 1.0 * r_11 + 6.12323399573677e-17 * r_12;
        let q6_x13 = q6_x11 * q6_x2 - q6_x12 * q6_x4
            + 6.12323399573677e-17 * r_21
            + 3.74939945665464e-33 * r_22;
        let q6_x14 = 1.0 * q6_x6;
        let q6_x15 = 1.0 * q6_x0;
        let q6 = libm::atan2(
            -6.12323399573677e-17 * q6_x10 * q6_x6 - q6_x13 * q6_x14
                + q6_x5 * (6.12323399573677e-17 * q6_x0 - 6.12323399573677e-17),
            q6_x10 * (-q6_x15 - 7.49879891330929e-33)
                + q6_x13 * (1.22464679914735e-16 * q6_x0 - 6.12323399573677e-17)
                + q6_x14 * q6_x5
                - (-q6_x15 - 3.74939945665464e-33)
                    * (-6.12323399573677e-17 * q6_x1 * q6_x11
                        + 1.0 * q6_x1 * q6_x8
                        + 6.12323399573677e-17 * q6_x12 * q6_x3
                        - q6_x7
                        - q6_x9),
        );

        // d_inner vector.
        let dinr_x0 = libm::sin(theta14);
        let dinr_x1 = libm::sin(q5);
        let dinr_x2 = libm::cos(theta14);
        let dinr_x3 = libm::cos(q5);
        let dinr_x4 = 1.22464679914735e-16 - 1.22464679914735e-16 * dinr_x3;
        let dinr_x5 = 1.0 * dinr_x3 + 1.49975978266186e-32;
        let dinr_x6 = libm::sin(q1);
        let dinr_x7 = p_y - 1.64748849439063e-19 * r_11 - 0.0823 * r_12;
        let dinr_x8 = libm::cos(q1);
        let dinr_x9 = p_x - 1.64748849439063e-19 * r_01 - 0.0823 * r_02;
        let dinr_x10 = 6.12323399573677e-17 * dinr_x2 - 6.12323399573677e-17;
        let dinr_x11 = 1.38777878078145e-17 * dinr_x4;
        let d_inner_x = -1.96734287847793e-17 * dinr_x0 * dinr_x4
            - 8.49769420904307e-34 * dinr_x0 * dinr_x5
            - 0.09465 * dinr_x0
            + 2.28650585388476e-18 * dinr_x1 * dinr_x2
            + 1.0 * dinr_x6 * dinr_x7
            + dinr_x8 * dinr_x9;
        let d_inner_y = 1.40008103759583e-34 * dinr_x0 * dinr_x1
            + dinr_x10 * dinr_x11
            + 5.79564097696485e-18 * dinr_x2
            + 2.28650585388476e-18 * dinr_x3
            - 1.0 * dinr_x6 * dinr_x9
            + dinr_x7 * dinr_x8
            + 0.10915;
        let d_inner_z = 2.28650585388476e-18 * dinr_x0 * dinr_x1
            + 0.09465 * dinr_x10 * dinr_x4
            + 1.38777878078145e-17 * dinr_x10 * dinr_x5
            + dinr_x11 * (1.0 * dinr_x2 + 3.74939945665464e-33)
            + 0.09465 * dinr_x2
            + 1.0 * p_z
            - 1.64748849439063e-19 * r_21
            - 0.0823 * r_22
            - 0.089159;
        let d_elbow =
            libm::sqrt(d_inner_x * d_inner_x + d_inner_y * d_inner_y + d_inner_z * d_inner_z);

        // SP3 for q3 (elbow).
        let q3_r_sq = 0.0277909737890625_f64;
        let q3_rhs = 0.16724253125 - 0.5 * d_elbow * d_elbow;
        let q3_phi = PI;
        let theta_q3_plus;
        let theta_q3_minus;
        if q3_r_sq < policy.axis_parallel * policy.axis_parallel {
            theta_q3_plus = 0.0;
            theta_q3_minus = 0.0;
        } else {
            let q3_r = libm::sqrt(q3_r_sq);
            if libm::fabs(q3_rhs) > q3_r + policy.subproblem_feasibility {
                let p = if q3_rhs > 0.0 { q3_phi } else { q3_phi + PI };
                theta_q3_plus = p;
                theta_q3_minus = p;
            } else {
                let clipped = (q3_rhs / q3_r).clamp(-1.0, 1.0);
                let delta = libm::acos(clipped);
                theta_q3_plus = q3_phi + delta;
                theta_q3_minus = q3_phi - delta;
            }
        }

        for q3 in [theta_q3_plus, theta_q3_minus] {
            let q2_x0 = libm::sin(q3);
            let q2_x1 = 0.39225 * q2_x0;
            let q2_x2 = libm::cos(q3);
            let q2_x3 = -0.39225 * q2_x2 - 0.425;
            let q2 = libm::atan2(
                d_inner_x * q2_x1
                    + d_inner_y * (-2.40183853482775e-17 * q2_x2 - 2.60237444818813e-17)
                    + d_inner_z * q2_x3,
                d_inner_x * q2_x3 - 2.40183853482775e-17 * d_inner_y * q2_x0 - d_inner_z * q2_x1,
            );
            let q4 = libm::remainder(theta14 - q2 - q3, TWO_PI);

            if count < 8 {
                out[count] = [q1, q2, q3, q4, q5, q6];
                count += 1;
            }
        }
    }

    count
}

/// Product-of-exponentials forward kinematics — mirrors `ur5_ik.fk`.
///
/// `q` is a 6-element joint-angle vector in radians. Returns the 4x4 SE(3)
/// pose of `ee_link` in `base_link`.
///
/// Delegates to the SymPy-generated, branch-free symbolic FK in
/// [`generated_kinematics::fk`] (regenerate with `codegen/generate_dynamics.py`).
pub fn fk(q: &[f64; 6]) -> SMatrix<f64, 4, 4> {
    let acc = super::generated_kinematics::fk(q);
    let mut out = SMatrix::<f64, 4, 4>::zeros();
    for i in 0..4 {
        for j in 0..4 {
            out[(i, j)] = acc[i][j];
        }
    }
    out
}

/// Frobenius-norm residual between FK(q) and the target pose.
fn fk_residual(q: &[f64; 6], target: &SMatrix<f64, 4, 4>) -> f64 {
    mat4_frob_diff(&fk(q), target)
}

--- generate_test_data.py ---
"""Generate the Rust reference dataset for the UR5 IK port.

Provenance for ``src/drivers/ur5/test_data.rs``. Reproduces the test data
required by the issue:

  1. Sample 500 valid SE(3) target poses (forward kinematics of random,
     in-limit joint configurations, guaranteeing reachability).
  2. Run the reference Python solver
     ``ssik.prebuilt.ur5_ik.solve(T, respect_limits=False)`` on each pose.
  3. Retain poses with >1 solution and emit ``(pose, solutions)`` pairs as a
     Rust ``static`` slice consumed by the ``#[cfg(test)]`` module.

Usage::

    pip install ssik
    python codegen/generate_test_data.py   # writes src/drivers/ur5/test_data.rs

The RNG seed is fixed so regenerating the file is deterministic.
"""

import io
import os

import numpy as np
from ssik.prebuilt import ur5_ik

rng = np.random.default_rng(12345)
# joint limits from artifact
lims = [(-6.28318530717959,6.28318530717959),
        (-6.28318530717959,6.28318530717959),
        (-3.14159265358979,3.14159265358979),
        (-6.28318530717959,6.28318530717959),
        (-6.28318530717959,6.28318530717959),
        (-6.28318530717959,6.28318530717959)]

cases=[]
attempts=0
while len(cases) < 500 and attempts < 20000:
    attempts+=1
    q = np.array([rng.uniform(lo,hi) for lo,hi in lims])
    T = ur5_ik.fk(q)
    sols = ur5_ik.solve(T, respect_limits=False)
    if len(sols) > 1:
        qs = [list(map(float, s.q)) for s in sols]
        cases.append((T.reshape(-1).tolist(), qs))

print("attempts",attempts,"cases",len(cases))
ncounts={}
for _,qs in cases:
    ncounts[len(qs)]=ncounts.get(len(qs),0)+1
print("solution-count distribution:",ncounts)

# write rust test data file
out=io.StringIO()
out.write("// AUTO-GENERATED test data. Do not edit by hand.\n")
out.write("// Generated by codegen/generate_test_data.py from\n")
out.write("// ssik.prebuilt.ur5_ik.solve(T, respect_limits=False).\n")
out.write("// Each case: (4x4 row-major target_tf, &[expected q solutions]).\n")
out.write("pub static CASES: &[(&[f64; 16], &[[f64; 6]])] = &[\n")
for tf, qs in cases:
    out.write("    (&[")
    out.write(", ".join(repr(float(x)) for x in tf))
    out.write("], &[")
    for q in qs:
        out.write("[" + ", ".join(repr(float(x)) for x in q) + "], ")
    out.write("]),\n")
out.write("];\n")
data=out.getvalue()
dest=os.path.join(os.path.dirname(__file__), "..", "src", "drivers", "ur5", "test_data.rs")
dest=os.path.normpath(dest)
with open(dest,"w") as f:
    f.write(data)
print("wrote", dest, "bytes", len(data))

def _build_kb() -> KinBody:
    """Reconstruct the baked KinBody. Run once at module import."""
    links = [Link(name=n) for n in _LINK_NAMES]
    joints = [
        Joint(
            name=_JOINT_NAMES[i],
            dof_index=i,
            parent_link=links[i],
            T_left=_JOINT_T_LEFTS[i],
            T_right=_JOINT_T_RIGHTS[i],
            axis=_JOINT_AXES[i],
            joint_type=_JOINT_TYPES[i],
            limits=_JOINT_LIMITS[i],
        )
        for i in range(len(_JOINT_NAMES))
    ]
    return KinBody(links=links, joints=joints)

def _solve_algebraic(T_target):
    """Algebraic IK candidates. Calls runtime SP6 for (q1, q5);
    inlines the post-SP6 SP1+SP3+SP1 chain.
    """
    r_00 = T_target[0, 0]
    r_01 = T_target[0, 1]
    r_02 = T_target[0, 2]
    r_10 = T_target[1, 0]
    r_11 = T_target[1, 1]
    r_12 = T_target[1, 2]
    r_20 = T_target[2, 0]
    r_21 = T_target[2, 1]
    r_22 = T_target[2, 2]
    p_x = T_target[0, 3]
    p_y = T_target[1, 3]
    p_z = T_target[2, 3]
    candidates = []

    # T_target-derived SP6 inputs.
    p_16_x = p_x - 1.64748849439063e-19*r_01 - 0.0823*r_02
    p_16_y = p_y - 1.64748849439063e-19*r_11 - 0.0823*r_12
    p_16_z = p_z - 1.64748849439063e-19*r_21 - 0.0823*r_22
    r_06_axes5_x = 1.0*r_02
    r_06_axes5_y = 1.0*r_12
    r_06_axes5_z = 1.0*r_22
    _H_SP_0 = np.array([0.0, -1.0, 6.123233995736766e-17])
    _K_SP_0 = np.array([-0.0, -0.0, -1.0])
    _K_SP_1 = np.array([0.0, -1.2246467991473532e-16, -1.0])
    _NEG_P_5 = np.array([-0.0, 1.3877787807814457e-17, 0.09465])
    _NEG_AXES_5 = np.array([-0.0, 1.0, -6.123233995736766e-17])
    # Build SP6 input arrays. h_sp / k_sp constant per arm; p_sp[0],
    # p_sp[2] depend on T_target via the inlined components above.
    p_16 = np.array([p_16_x, p_16_y, p_16_z])
    r_06_axes5 = np.array([r_06_axes5_x, r_06_axes5_y, r_06_axes5_z])
    h_sp = (_H_SP_0, _H_SP_0, _H_SP_0, _H_SP_0)
    k_sp = (_K_SP_0, _K_SP_1, _K_SP_0, _K_SP_1)
    p_sp = (p_16, _NEG_P_5, r_06_axes5, _NEG_AXES_5)
    theta15_solutions, _ = _sp6_runtime.solve(h_sp, k_sp, p_sp, 0.10915, 0.0)

    for q1, q5 in theta15_solutions:
        s1 = math.sin(q1)
        c1 = math.cos(q1)
        s5 = math.sin(q5)
        c5 = math.cos(q5)
        # SP1 for theta14 = q1+q2+q3+q4 (sum of parallel-axis rotations).
        th14_x0 = math.sin(q5)
        th14_x1 = 1.0*r_22
        th14_x2 = math.sin(q1)
        th14_x3 = 6.12323399573677e-17*r_01 - 1.0*r_02
        th14_x4 = th14_x2*th14_x3
        th14_x5 = 1.0*r_01 + 6.12323399573677e-17*r_02
        th14_x6 = 6.12323399573677e-17*th14_x2*th14_x5
        th14_x7 = math.cos(q1)
        th14_x8 = 6.12323399573677e-17*r_11 - 1.0*r_12
        th14_x9 = th14_x7*th14_x8
        th14_x10 = 1.0*r_11 + 6.12323399573677e-17*r_12
        th14_x11 = 6.12323399573677e-17*th14_x10*th14_x7
        th14_x12 = th14_x11 + th14_x4 - th14_x6 - th14_x9
        th14_x13 = math.cos(q5)
        th14_x14 = 6.12323399573677e-17*th14_x10*th14_x2 - th14_x2*th14_x8 - th14_x3*th14_x7 + 6.12323399573677e-17*th14_x5*th14_x7
        th14_x15 = 1.0*th14_x13
        theta14 = math.atan2(-th14_x0*th14_x1 - 6.12323399573677e-17*th14_x0*th14_x12 + th14_x14*(6.12323399573677e-17 - 6.12323399573677e-17*th14_x13), -1.0*th14_x0*th14_x14 + th14_x1*(1.22464679914735e-16*th14_x13 - 6.12323399573677e-17) + th14_x12*(-th14_x15 - 7.49879891330929e-33) - (th14_x15 + 3.74939945665464e-33)*(6.12323399573677e-17*r_22 - th14_x11 - th14_x4 + th14_x6 + th14_x9))
        # SP1 for q6 (wrist roll-2): closed-form atan2.
        q6_x0 = math.cos(q5)
        q6_x1 = math.sin(q1)
        q6_x2 = 1.0*q6_x1
        q6_x3 = math.cos(q1)
        q6_x4 = 1.0*q6_x3
        q6_x5 = q6_x2*r_00 - q6_x4*r_10 + 6.12323399573677e-17*r_20
        q6_x6 = math.sin(q5)
        q6_x7 = 6.12323399573677e-17*r_22
        q6_x8 = 6.12323399573677e-17*r_01 - 1.0*r_02
        q6_x9 = q6_x4*(6.12323399573677e-17*r_11 - 1.0*r_12)
        q6_x10 = q6_x2*q6_x8 - q6_x7 - q6_x9 + 3.74939945665464e-33*r_21
        q6_x11 = 1.0*r_01 + 6.12323399573677e-17*r_02
        q6_x12 = 1.0*r_11 + 6.12323399573677e-17*r_12
        q6_x13 = q6_x11*q6_x2 - q6_x12*q6_x4 + 6.12323399573677e-17*r_21 + 3.74939945665464e-33*r_22
        q6_x14 = 1.0*q6_x6
        q6_x15 = 1.0*q6_x0
        q6 = math.atan2(-6.12323399573677e-17*q6_x10*q6_x6 - q6_x13*q6_x14 + q6_x5*(6.12323399573677e-17*q6_x0 - 6.12323399573677e-17), q6_x10*(-q6_x15 - 7.49879891330929e-33) + q6_x13*(1.22464679914735e-16*q6_x0 - 6.12323399573677e-17) + q6_x14*q6_x5 - (-q6_x15 - 3.74939945665464e-33)*(-6.12323399573677e-17*q6_x1*q6_x11 + 1.0*q6_x1*q6_x8 + 6.12323399573677e-17*q6_x12*q6_x3 - q6_x7 - q6_x9))
        s14 = math.sin(theta14)
        c14 = math.cos(theta14)
        # d_inner = r_01.T @ p_16 - p[1] - r_14 @ r_45 @ p[5] - r_14 @ p[4]
        dinr_x0 = math.sin(theta14)
        dinr_x1 = math.sin(q5)
        dinr_x2 = math.cos(theta14)
        dinr_x3 = math.cos(q5)
        dinr_x4 = 1.22464679914735e-16 - 1.22464679914735e-16*dinr_x3
        dinr_x5 = 1.0*dinr_x3 + 1.49975978266186e-32
        dinr_x6 = math.sin(q1)
        dinr_x7 = p_y - 1.64748849439063e-19*r_11 - 0.0823*r_12
        dinr_x8 = math.cos(q1)
        dinr_x9 = p_x - 1.64748849439063e-19*r_01 - 0.0823*r_02
        dinr_x10 = 6.12323399573677e-17*dinr_x2 - 6.12323399573677e-17
        dinr_x11 = 1.38777878078145e-17*dinr_x4
        d_inner_x = -1.96734287847793e-17*dinr_x0*dinr_x4 - 8.49769420904307e-34*dinr_x0*dinr_x5 - 0.09465*dinr_x0 + 2.28650585388476e-18*dinr_x1*dinr_x2 + 1.0*dinr_x6*dinr_x7 + dinr_x8*dinr_x9
        d_inner_y = 1.40008103759583e-34*dinr_x0*dinr_x1 + dinr_x10*dinr_x11 + 5.79564097696485e-18*dinr_x2 + 2.28650585388476e-18*dinr_x3 - 1.0*dinr_x6*dinr_x9 + dinr_x7*dinr_x8 + 0.10915
        d_inner_z = 2.28650585388476e-18*dinr_x0*dinr_x1 + 0.09465*dinr_x10*dinr_x4 + 1.38777878078145e-17*dinr_x10*dinr_x5 + dinr_x11*(1.0*dinr_x2 + 3.74939945665464e-33) + 0.09465*dinr_x2 + 1.0*p_z - 1.64748849439063e-19*r_21 - 0.0823*r_22 - 0.089159
        d_elbow = math.sqrt(d_inner_x*d_inner_x + d_inner_y*d_inner_y + d_inner_z*d_inner_z)
        # SP3 for q3 (elbow): reduces to SP4 with d_elbow target.
        _q3_R_sq = 0.0277909737890625
        _q3_rhs = 0.16724253125 - 1/2*d_elbow**2
        _q3_phi = math.pi
        if _q3_R_sq < _DEG_SQ:
            theta_q3_plus = 0.0
            theta_q3_minus = 0.0  # degenerate; verify-step drops
        else:
            _q3_R = math.sqrt(_q3_R_sq)
            if abs(_q3_rhs) > _q3_R + _FEAS_TOL:
                # LS fallback: theta = phi (or phi + pi if rhs < 0)
                theta_q3_plus = (
                    _q3_phi if _q3_rhs > 0 else _q3_phi + math.pi
                )
                theta_q3_minus = theta_q3_plus
            else:
                _q3_clipped = min(1.0, max(-1.0, _q3_rhs / _q3_R))
                _q3_delta = math.acos(_q3_clipped)
                theta_q3_plus = _q3_phi + _q3_delta
                theta_q3_minus = _q3_phi - _q3_delta

        for q3 in (theta_q3_plus, theta_q3_minus):
            s3 = math.sin(q3)
            c3 = math.cos(q3)
            # SP1 for q2 (shoulder pitch): closed-form atan2.
            q2_x0 = math.sin(q3)
            q2_x1 = 0.39225*q2_x0
            q2_x2 = math.cos(q3)
            q2_x3 = -0.39225*q2_x2 - 0.425
            q2 = math.atan2(d_inner_x*q2_x1 + d_inner_y*(-2.40183853482775e-17*q2_x2 - 2.60237444818813e-17) + d_inner_z*q2_x3, d_inner_x*q2_x3 - 2.40183853482775e-17*d_inner_y*q2_x0 - d_inner_z*q2_x1)
            q4 = ((theta14 - q2 - q3 + math.pi) % (2.0 * math.pi)) - math.pi
            candidates.append([q1, q2, q3, q4, q5, q6])
    return candidates

@cython.ccall

    b20=cython.double, b21=cython.double, b22=cython.double, b23=cython.double,
)
def _fk(q):
    """POE forward kinematics using the baked chain constants.

    Hand-rolled scalar 4x4 matmul + inline Rodrigues -- no per-call
    ``np.eye(4)`` allocations and no per-joint numpy ``@`` dispatch.
    Each numpy ``@`` on a 4x4 has ~3 us of dispatch overhead;
    inlining the ~85 scalar ops per joint turns the inner loop into
    a single native-code chunk under Cython compile.

    Bottom row of the accumulator stays [0, 0, 0, 1] implicitly.
    """
    n = len(_JOINT_AXES)
    # Identity accumulator (the bottom row [0,0,0,1] is implicit).
    a00 = 1.0; a01 = 0.0; a02 = 0.0; a03 = 0.0
    a10 = 0.0; a11 = 1.0; a12 = 0.0; a13 = 0.0
    a20 = 0.0; a21 = 0.0; a22 = 1.0; a23 = 0.0
    for i in range(n):
        # Inline Rodrigues for this joint's axis.
        ax = float(_JOINT_AXES[i][0])
        ay = float(_JOINT_AXES[i][1])
        az = float(_JOINT_AXES[i][2])
        qi = float(q[i])
        c = math.cos(qi); s = math.sin(qi); oc = 1.0 - c
        r00 = c + ax*ax*oc;     r01 = ax*ay*oc - az*s; r02 = ax*az*oc + ay*s
        r10 = ay*ax*oc + az*s;  r11 = c + ay*ay*oc;    r12 = ay*az*oc - ax*s
        r20 = az*ax*oc - ay*s;  r21 = az*ay*oc + ax*s; r22 = c + az*az*oc
        # T_left[i] entries.
        Tl = _JOINT_T_LEFTS[i]
        l00 = float(Tl[0,0]); l01 = float(Tl[0,1])
        l02 = float(Tl[0,2]); l03 = float(Tl[0,3])
        l10 = float(Tl[1,0]); l11 = float(Tl[1,1])
        l12 = float(Tl[1,2]); l13 = float(Tl[1,3])
        l20 = float(Tl[2,0]); l21 = float(Tl[2,1])
        l22 = float(Tl[2,2]); l23 = float(Tl[2,3])
        # M = T_left[i] @ R (R is the homogeneous version of the 3x3
        # rotation above with column 3 = [0,0,0,1]^T).
        m00 = l00*r00 + l01*r10 + l02*r20
        m01 = l00*r01 + l01*r11 + l02*r21
        m02 = l00*r02 + l01*r12 + l02*r22
        m03 = l03
        m10 = l10*r00 + l11*r10 + l12*r20
        m11 = l10*r01 + l11*r11 + l12*r21
        m12 = l10*r02 + l11*r12 + l12*r22
        m13 = l13
        m20 = l20*r00 + l21*r10 + l22*r20
        m21 = l20*r01 + l21*r11 + l22*r21
        m22 = l20*r02 + l21*r12 + l22*r22
        m23 = l23
        # T_right[i] entries.
        Tr = _JOINT_T_RIGHTS[i]
        t00 = float(Tr[0,0]); t01 = float(Tr[0,1])
        t02 = float(Tr[0,2]); t03 = float(Tr[0,3])
        t10 = float(Tr[1,0]); t11 = float(Tr[1,1])
        t12 = float(Tr[1,2]); t13 = float(Tr[1,3])
        t20 = float(Tr[2,0]); t21 = float(Tr[2,1])
        t22 = float(Tr[2,2]); t23 = float(Tr[2,3])
        # N = M @ T_right[i]
        n00 = m00*t00 + m01*t10 + m02*t20
        n01 = m00*t01 + m01*t11 + m02*t21
        n02 = m00*t02 + m01*t12 + m02*t22
        n03 = m00*t03 + m01*t13 + m02*t23 + m03
        n10 = m10*t00 + m11*t10 + m12*t20
        n11 = m10*t01 + m11*t11 + m12*t21
        n12 = m10*t02 + m11*t12 + m12*t22
        n13 = m10*t03 + m11*t13 + m12*t23 + m13
        n20 = m20*t00 + m21*t10 + m22*t20
        n21 = m20*t01 + m21*t11 + m22*t21
        n22 = m20*t02 + m21*t12 + m22*t22
        n23 = m20*t03 + m21*t13 + m22*t23 + m23
        # T_acc = T_acc @ N
        b00 = a00*n00 + a01*n10 + a02*n20
        b01 = a00*n01 + a01*n11 + a02*n21
        b02 = a00*n02 + a01*n12 + a02*n22
        b03 = a00*n03 + a01*n13 + a02*n23 + a03
        b10 = a10*n00 + a11*n10 + a12*n20
        b11 = a10*n01 + a11*n11 + a12*n21
        b12 = a10*n02 + a11*n12 + a12*n22
        b13 = a10*n03 + a11*n13 + a12*n23 + a13
        b20 = a20*n00 + a21*n10 + a22*n20
        b21 = a20*n01 + a21*n11 + a22*n21
        b22 = a20*n02 + a21*n12 + a22*n22
        b23 = a20*n03 + a21*n13 + a22*n23 + a23
        a00, a01, a02, a03 = b00, b01, b02, b03
        a10, a11, a12, a13 = b10, b11, b12, b13
        a20, a21, a22, a23 = b20, b21, b22, b23
    return np.array(
        [[a00, a01, a02, a03],
         [a10, a11, a12, a13],
         [a20, a21, a22, a23],
         [0.0, 0.0, 0.0, 1.0]],
        dtype=np.float64,
    )


@cython.ccall
@cython.locals(i=cython.int, n=cython.int)
def _spatial_jacobian(q):
    """6 x n_dof spatial Jacobian using the baked chain constants.

    Math identical to ssik.refinement.kinbody_jacobian: column i
    is (p_i x z_i, z_i) where z_i is the i-th joint axis in the
    world frame at q and p_i is the i-th joint origin. This is
    the SPATIAL twist representation -- T(q+dq) @ T(q)^-1 ~
    exp([J @ dq]) -- matching the residual extracted by
    ssik.refinement.se3_log_residual. Per-arm version with
    baked _JOINT_AXES / _JOINT_T_LEFTS / _JOINT_T_RIGHTS so
    there's no KinBody walk at runtime.
    """
    n = len(_JOINT_AXES)
    cum = _FK_EYE4.copy()
    cums = [cum.copy()]
    rot = _FK_EYE4.copy()
    for i in range(n):
        rot[:3, :3] = _rotation_matrix(_JOINT_AXES[i], float(q[i]))
        cum = cum @ _JOINT_T_LEFTS[i] @ rot @ _JOINT_T_RIGHTS[i]
        cums.append(cum.copy())
    J = np.zeros((6, n), dtype=np.float64)
    for i in range(n):
        t_pre = cums[i] @ _JOINT_T_LEFTS[i]
        axis_unit = _JOINT_AXES[i] / np.linalg.norm(_JOINT_AXES[i])
        z_i = t_pre[:3, :3] @ axis_unit
        p_i = t_pre[:3, 3]
        J[:3, i] = np.cross(p_i, z_i)
        J[3:, i] = z_i
    return J


@cython.ccall
def _wrap_to_pi(a: float) -> float:
    """Wrap an angle to ``(-pi, pi]``. Called inside the per-IK
    dedup hot loop (235k+ times on Franka 7R)."""
    return ((a + math.pi) % _TWO_PI) - math.pi


@cython.ccall
@cython.locals(
    i=cython.int,
    n=cython.int,
    diff=cython.double,
    ai=cython.double,
    bi=cython.double,
)
def _q_close_wrap(a, b, tol: float) -> bool:
    """Return ``True`` if joint vectors ``a`` and ``b`` agree (mod 2pi)
    within ``tol`` per element. Replaces the
    ``np.array([_wrap_to_pi(...)]) -> np.all(np.abs(...) < tol)``
    pipeline that allocated a numpy array per dedup-loop iteration --
    a per-element scalar loop avoids the array creation and the
    ``np.all`` reduction overhead, which together dominated the
    artifact's ``solve()`` body at the per-IK level."""
    n = len(a)
    for i in range(n):
        ai = float(a[i])
        bi = float(b[i])
        diff = ((ai - bi + math.pi) % _TWO_PI) - math.pi
        if abs(diff) > tol:
            return False
    return True


def solve(
    T_target,
    *,
    max_solutions: int | None = None,
    q_seed=None,
    respect_limits: bool = True,
    allow_refinement: bool = False,
    policy: TolerancePolicy = DEFAULT_TOLERANCE_POLICY,
    refinement_max_iters: int = 15,
):
    """Inverse kinematics. Returns ``list[Solution]``.

    :param T_target: 4x4 SE(3) target end-effector pose.
    :param max_solutions: optional cap on returned IKs (post-dedup,
        post-limits filter). ``None`` = full redundancy enumeration.
        Combine with ``q_seed`` for the "give me the IK closest to
        where I am now" trajectory-tracking idiom.
    :param q_seed: optional joint configuration. When provided,
        returned solutions are sorted by wrap-to-pi distance from
        ``q_seed`` (closest first). Has no effect on which solutions
        are returned, only their order.
    :param respect_limits: when ``True`` (default), solutions
        outside URDF joint limits are dropped. Pass ``False`` for
        the raw geometric set (e.g. analysis / debugging).
    :param allow_refinement: opt into Newton polish for near-miss
        algebraic candidates that don't quite meet ``fk_atol``.
        Default ``False`` -- the algebraic path is already at
        machine precision on tier-0 / SRS arms. On tier-2 RR
        arms (JACO 2, Rizon 4, Kassow), polish can recover
        edge-case candidates whose algebraic FK drifts above
        ``fk_atol``, at ~100-300 us per polished branch.
    :param policy: tolerance policy (FK closure + dedup tolerance).
        Rarely customised.
    :param refinement_max_iters: cap on Newton iterations per
        candidate when ``allow_refinement=True``.
    :returns: list of :class:`Solution`; empty list iff no IK
        closed within ``policy.subproblem_numerical`` (or all
        IKs were filtered by ``respect_limits=True``).
    """
    T = np.asarray(T_target, dtype=np.float64)
    candidates = _solve_algebraic(T)

    fk_atol = policy.subproblem_numerical
    dedup_atol = policy.subproblem_dedup

    # Three-bucket sort: exact (closes within fk_atol), near-miss
    # (refinable when allow_refinement=True), or drop.
    verified: list[tuple[np.ndarray, float, str, int]] = []
    for cand_q in candidates:
        q = np.asarray(cand_q, dtype=np.float64)
        if not np.all(np.isfinite(q)):
            continue
        T_check = _fk(q)
        residual = float(np.linalg.norm(T_check - T))
        if residual <= fk_atol:
            verified.append((q, residual, "none", 0))
            continue
        if not allow_refinement:
            continue
        # Newton polish using the per-arm spatial Jacobian.
        refined = _lm_refine(
            q,
            _fk,
            T,
            fk_atol=fk_atol,
            max_iters=refinement_max_iters,
            jacobian_fn=_spatial_jacobian,
        )
        if refined is None:
            continue
        q_ref, resid_ref, iters = refined
        verified.append((q_ref, resid_ref, "lm", iters))

    # Wrap-to-pi dedup; keep lowest fk_residual on collision.
    # Inner check via ``_q_close_wrap`` -- typed scalar loop, no per-
    # iteration numpy allocation (#137 Slice 3).
    deduped: list[tuple[np.ndarray, float, str, int]] = []
    for cand_q, cand_res, ref_used, ref_iters in verified:
        dup_idx = None
        for j, (existing_q, _, _, _) in enumerate(deduped):
            if _q_close_wrap(cand_q, existing_q, dedup_atol):
                dup_idx = j
                break
        if dup_idx is None:
            deduped.append((cand_q, cand_res, ref_used, ref_iters))
        elif cand_res < deduped[dup_idx][1]:
            deduped[dup_idx] = (cand_q, cand_res, ref_used, ref_iters)

    solutions = [
        Solution(
            q=q,
            fk_residual=residual,
            refinement_used=ref_used,
        )
        for q, residual, ref_used, _ref_iters in deduped
    ]

    # Post-processing pass (#238 item 4). Order matters:
    #   1. wrap_to_limits tries q +/- 2*pi per joint to bring
    #      candidates into the URDF's limit range
    #   2. respect_limits drops anything still outside
    #   3. nearest_to_seed sorts by distance to q_seed (if given)
    #   4. max_solutions truncates to the first k
    if respect_limits:
        solutions = _ps_wrap_to_limits(solutions, _KB)
        solutions = _ps_respect_limits(solutions, _KB)
    if q_seed is not None:
        solutions = _ps_nearest_to_seed(solutions, q_seed)
    if max_solutions is not None and len(solutions) > max_solutions:
        solutions = solutions[:max_solutions]
    return solutions
